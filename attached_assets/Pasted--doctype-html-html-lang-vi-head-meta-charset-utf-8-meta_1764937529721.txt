<script>
/*
 Ultimate Gomoku - HTML/Canvas port
 Uses decimal.js for camera/zoom math; apply toFixed(8) when converting to Number for drawing.
*/

// ---------- Config ----------
const canvas = document.getElementById('c');
const ctx = canvas.getContext('2d');

let WIDTH = window.innerWidth;
let HEIGHT = window.innerHeight - document.getElementById('topbar').offsetHeight;
canvas.width = WIDTH;
canvas.height = HEIGHT;

const CELL_BASE = 48;            // base pixel per cell at zoom 1
let ZOOM = new Decimal(1);       // Decimal
const ZOOM_MIN = new Decimal(0.1);
const ZOOM_MAX = new Decimal(5);

const ARROW_MIN = 10;
const ARROW_MAX = 150;
const ARROW_SCALE_DIST = 20.0;

const COLORS = {
  grid: '#c8c8c8',
  bg: '#ffffff',
  X: '#dc3737',
  O: '#2d63e0',
  hover: '#50c970',
  win: '#0aa050',
  uiBg: '#ebebeb'
};

// ---------- Game state ----------
let board = new Map(); // key "gx,gy" -> 'X'/'O'
let starter = 'X';
let current = starter;
let last_winner = null;
let score = { X: 0, O: 0 };

let camera_x = new Decimal(WIDTH / 2);   // pixel of grid origin (top-left of cell (0,0))
let camera_y = new Decimal(HEIGHT / 2);

let dragging = false;
let dragStart = null;

let winner_line = [];  // [ {gx,gy}, {gx,gy} ] endpoints
let game_over = false;

let arrow_hit_map = new Map(); // key "gx,gy" -> {x,y,r}

// DOM elements
const scoreXEl = document.getElementById('scoreX');
const scoreOEl = document.getElementById('scoreO');
const newGameBtn = document.getElementById('newGame');
const zoomLabel = document.getElementById('zoomLabel');
const turnDisplay = document.getElementById('turnDisplay');

// Decimal default: we will simply use Decimal arithmetic and call toFixed(8) when converting to Number.
// (decimal.set precision is for significant digits; we rely on toFixed(8) to convert)
Decimal.set({ toExpNeg: -9e15, toExpPos: 9e15 }); // avoid exponential in usual range

// ---------- Helpers ----------
function cellSize() {
  return ZOOM.mul(CELL_BASE); // Decimal
}
function gridKey(gx, gy){ return gx + ',' + gy; }

function gridToScreenTopLeftDec(gx, gy){
  // returns {x: Decimal, y: Decimal}
  const s = cellSize();
  const x = new Decimal(gx).mul(s).plus(camera_x);
  const y = new Decimal(gy).mul(s).plus(camera_y);
  return {x, y};
}
function gridToScreenCenterDec(gx, gy){
  const tl = gridToScreenTopLeftDec(gx, gy);
  const s = cellSize();
  return { x: tl.x.plus(s.div(2)), y: tl.y.plus(s.div(2)) };
}
function screenToGridDec(mx, my){
  // Decimal floor
  const s = cellSize();
  const gx = Decimal.floor(new Decimal(mx).minus(camera_x).div(s));
  const gy = Decimal.floor(new Decimal(my).minus(camera_y).div(s));
  return { gx: gx.toNumber(), gy: gy.toNumber() }; // grid indices small enough for Number
}

function clamp(v, a, b){ return Math.max(a, Math.min(b, v)); }

// ---------- Win detection ----------
const directions = [[1,0],[0,1],[1,1],[1,-1]];
function checkWinAndLine(gx, gy, player){
  for (let d=0; d<directions.length; d++){
    const [dx,dy] = directions[d];
    let line = [[gx,gy]];
    // forward
    let cx = gx+dx, cy = gy+dy;
    while(board.get(gridKey(cx,cy)) === player){ line.push([cx,cy]); cx += dx; cy += dy; }
    // backward
    cx = gx-dx; cy = gy-dy;
    while(board.get(gridKey(cx,cy)) === player){ line.push([cx,cy]); cx -= dx; cy -= dy; }
    if (line.length >= 5){
      // find endpoints along projection
      line.sort((a,b) => (a[0]*dx + a[1]*dy) - (b[0]*dx + b[1]*dy));
      winner_line = [ {gx: line[0][0], gy: line[0][1]}, {gx: line[line.length-1][0], gy: line[line.length-1][1]} ];
      return true;
    }
  }
  return false;
}

// ---------- Drawing (use toFixed(8) when converting Decimal -> Number) ----------
function toNum(dec){ // dec is Decimal -> number with 8 dp
  // sometimes dec is number already
  if (Decimal.isDecimal(dec)) return Number(dec.toFixed(8));
  return Number(new Decimal(dec).toFixed(8));
}

function drawGrid(){
  ctx.fillStyle = COLORS.bg;
  ctx.fillRect(0,0,WIDTH,HEIGHT);
  ctx.strokeStyle = COLORS.grid;
  ctx.lineWidth = 1;

  const sDec = cellSize();
  const s = toNum(sDec);
  // visible range
  const gx0 = Math.floor( toNum( Decimal.floor( new Decimal(-camera_x).minus(sDec).div(sDec) ) ) ) - 1;
  const gx1 = Math.floor( toNum( Decimal.floor( new Decimal(WIDTH).minus(camera_x).div(sDec) ) ) ) + 1;
  const gy0 = Math.floor( toNum( Decimal.floor( new Decimal(-camera_y).minus(sDec).div(sDec) ) ) ) - 1;
  const gy1 = Math.floor( toNum( Decimal.floor( new Decimal(HEIGHT).minus(camera_y).div(sDec) ) ) ) + 1;

  ctx.beginPath();
  for (let gx = gx0; gx <= gx1; gx++){
    const x = toNum( new Decimal(gx).mul(sDec).plus(camera_x) );
    ctx.moveTo(x, 0); ctx.lineTo(x, HEIGHT);
  }
  for (let gy = gy0; gy <= gy1; gy++){
    const y = toNum( new Decimal(gy).mul(sDec).plus(camera_y) );
    ctx.moveTo(0, y); ctx.lineTo(WIDTH, y);
  }
  ctx.stroke();
}

function drawPieces(){
  const s = toNum(cellSize());
  for (const [key, p] of board.entries()){
    const [gx,gy] = key.split(',').map(Number);
    const center = gridToScreenCenterDec(gx, gy);
    const cx = toNum(center.x), cy = toNum(center.y);
    // skip extremely far to keep perf
    if (cx < -2000 || cx > WIDTH + 2000 || cy < -2000 || cy > HEIGHT + 2000) continue;

    if (p === 'X'){
      ctx.strokeStyle = COLORS.X;
      ctx.lineWidth = Math.max(1, Math.round(3 * toNum(ZOOM)));
      const half = Math.max(6, (s/2) * 0.45);
      ctx.beginPath();
      ctx.moveTo(cx - half, cy - half);
      ctx.lineTo(cx + half, cy + half);
      ctx.moveTo(cx - half, cy + half);
      ctx.lineTo(cx + half, cy - half);
      ctx.stroke();
    } else {
      ctx.strokeStyle = COLORS.O;
      ctx.lineWidth = Math.max(1, Math.round(3 * toNum(ZOOM)));
      const radius = Math.max(4, Math.round((s/2) * 0.45));
      ctx.beginPath();
      ctx.arc(cx, cy, radius, 0, Math.PI*2);
      ctx.stroke();
    }
  }
}

function drawHover(mx, my){
  if (game_over) return;
  const {gx,gy} = screenToGridDec(mx, my);
  const tl = gridToScreenTopLeftDec(gx, gy);
  const x = toNum(tl.x), y = toNum(tl.y);
  const s = toNum(cellSize());
  ctx.strokeStyle = COLORS.hover;
  ctx.lineWidth = Math.max(1, Math.round(3 * toNum(ZOOM)));
  ctx.strokeRect(x+1, y+1, s-2, s-2);
}

function drawWinnerLine(){
  if (!winner_line || winner_line.length < 2) return;
  const p1 = gridToScreenCenterDec(winner_line[0].gx, winner_line[0].gy);
  const p2 = gridToScreenCenterDec(winner_line[1].gx, winner_line[1].gy);
  const x1 = toNum(p1.x), y1 = toNum(p1.y);
  const x2 = toNum(p2.x), y2 = toNum(p2.y);
  ctx.strokeStyle = COLORS.win;
  ctx.lineWidth = Math.max(3, Math.round(6 * toNum(ZOOM)));
  ctx.beginPath();
  ctx.moveTo(x1, y1);
  ctx.lineTo(x2, y2);
  ctx.stroke();
}

// ---------- Arrows ----------
function computeAndDrawArrows(){
  arrow_hit_map.clear();
  const cx = WIDTH/2, cy = HEIGHT/2;
  const s = toNum(cellSize());
  for (const [key, player] of board.entries()){
    const [gx,gy] = key.split(',').map(Number);
    const center = gridToScreenCenterDec(gx, gy);
    const px = toNum(center.x), py = toNum(center.y);
    if (px >= 0 && px <= WIDTH && py >= 0 && py <= HEIGHT) continue; // visible
    const dx = px - cx, dy = py - cy;
    const dist = Math.hypot(dx, dy);
    if (dist === 0) continue;
    const angle = Math.atan2(dy, dx);
    // intersect with screen rect
    let tCandidates = [];
    if (dx !== 0){
      let t = (0 - cx) / dx; if (t>0) tCandidates.push(t);
      t = (WIDTH - cx) / dx; if (t>0) tCandidates.push(t);
    }
    if (dy !== 0){
      let t = (0 - cy) / dy; if (t>0) tCandidates.push(t);
      t = (HEIGHT - cy) / dy; if (t>0) tCandidates.push(t);
    }
    if (tCandidates.length === 0) continue;
    const tmin = Math.min(...tCandidates);
    const ex = cx + dx * tmin;
    const ey = cy + dy * tmin;
    // size based on distance in cells
    const cellDist = dist / Math.max(1.0, s);
    let size = ARROW_MAX * (1.0 / (1.0 + (cellDist / ARROW_SCALE_DIST)));
    size = clamp(size, ARROW_MIN, ARROW_MAX);
    const tip = {x: ex, y: ey};
    const left = {x: ex - size * Math.cos(angle - 0.35), y: ey - size * Math.sin(angle - 0.35)};
    const right = {x: ex - size * Math.cos(angle + 0.35), y: ey - size * Math.sin(angle + 0.35)};
    ctx.fillStyle = (player === 'X') ? COLORS.X : COLORS.O;
    ctx.beginPath();
    ctx.moveTo(tip.x, tip.y);
    ctx.lineTo(left.x, left.y);
    ctx.lineTo(right.x, right.y);
    ctx.closePath();
    ctx.fill();
    const hitRadius = Math.max(8, Math.round(size * 0.45));
    arrow_hit_map.set(key, {x: tip.x, y: tip.y, r: hitRadius});
  }
}

// ---------- UI updates ----------
function updateUI(){
  scoreXEl.textContent = `X : ${score.X}`;
  scoreOEl.textContent = `O : ${score.O}`;
  zoomLabel.textContent = `${toNum(ZOOM).toFixed(2)}x`;
  // turn display bottom-left
  turnDisplay.style.background = 'rgba(255,255,255,0.9)';
  turnDisplay.style.color = (current === 'X') ? COLORS.X : COLORS.O;
  turnDisplay.style.padding = '6px 8px';
  turnDisplay.style.borderRadius = '6px';
  turnDisplay.textContent = `${current} turns`;
}

// ---------- New game ----------
function newGame(byWin=null){
  board.clear();
  winner_line = [];
  game_over = false;
  if (byWin === 'X' || byWin === 'O'){
    // loser starts next
    const loser = (byWin === 'X') ? 'O' : 'X';
    starter = loser;
    last_winner = byWin;
  } else {
    last_winner = null;
    // keep starter the same
  }
  current = starter;
}

// ---------- Input handling ----------
canvas.addEventListener('contextmenu', e => { e.preventDefault(); }); // disable context menu
let isRightDown = false;
canvas.addEventListener('mousedown', (e) => {
  if (e.button === 2){ // right
    isRightDown = true;
    dragging = true;
    dragStart = {x: e.clientX, y: e.clientY};
  } else if (e.button === 0){ // left
    const rect = canvas.getBoundingClientRect();
    const mx = e.clientX - rect.left;
    const my = e.clientY - rect.top;
    // check UI New Game click? (we handle via button)
    // check arrow hit first
    let clickedArrow = null;
    for (const [key, val] of arrow_hit_map.entries()){
      const dx = mx - val.x, dy = my - val.y;
      if (dx*dx + dy*dy <= val.r * val.r){ clickedArrow = key; break; }
    }
    if (clickedArrow){
      // teleport
      const [gx,gy] = clickedArrow.split(',').map(Number);
      const s = cellSize();
      // center that cell
      camera_x = new Decimal(WIDTH/2).minus( new Decimal(gx).mul(s).plus(s.div(2)) );
      camera_y = new Decimal(HEIGHT/2).minus( new Decimal(gy).mul(s).plus(s.div(2)) );
      // do NOT place a piece
      return;
    }
    if (!game_over){
      const {gx,gy} = screenToGridDec(mx, my);
      if (!board.has(gridKey(gx,gy))){
        board.set(gridKey(gx,gy), current);
        if (checkWinAndLine(gx, gy, current)){
          score[current] += 1;
          game_over = true;
          // set starter to loser
          const loser = (current === 'X') ? 'O' : 'X';
          starter = loser;
          last_winner = current;
        }
        if (!game_over) current = (current === 'X') ? 'O' : 'X';
      }
    }
  }
});
window.addEventListener('mouseup', (e) => {
  if (e.button === 2){ isRightDown = false; dragging = false; }
});
window.addEventListener('mousemove', (e) => {
  if (dragging && isRightDown){
    // pan by delta
    if (dragStart){
      const dx = e.clientX - dragStart.x;
      const dy = e.clientY - dragStart.y;
      camera_x = camera_x.plus(dx);
      camera_y = camera_y.plus(dy);
      dragStart = {x: e.clientX, y: e.clientY};
    }
  }
});

canvas.addEventListener('wheel', (e) => {
  e.preventDefault();
  const rect = canvas.getBoundingClientRect();
  const mx = e.clientX - rect.left;
  const my = e.clientY - rect.top;
  const delta = -Math.sign(e.deltaY);
  const factor = 1 + 0.12 * delta;
  const oldZoom = ZOOM;
  ZOOM = Decimal.max(ZOOM_MIN, Decimal.min(ZOOM_MAX, ZOOM.mul(factor)));
  // keep mouse point stable: camera_new = mx - ((mx - camera_old) / oldZoom) * newZoom
  camera_x = new Decimal(mx).minus( new Decimal(mx).minus(camera_x).div(oldZoom).mul(ZOOM) );
  camera_y = new Decimal(my).minus( new Decimal(my).minus(camera_y).div(oldZoom).mul(ZOOM) );
  updateUI();
}, { passive: false });

newGameBtn.addEventListener('click', ()=> newGame(null));

// handle resize
window.addEventListener('resize', ()=>{
  WIDTH = window.innerWidth;
  HEIGHT = window.innerHeight - document.getElementById('topbar').offsetHeight;
  canvas.width = WIDTH;
  canvas.height = HEIGHT;
  updateUI();
});

// ---------- Main render loop ----------
function loop(){
  drawGrid();
  drawPieces();
  // mouse pos for hover
  const rect = canvas.getBoundingClientRect();
  const mx = (window.event && window.event.clientX) ? window.event.clientX - rect.left : (canvas._lastMouseX || 0);
  const my = (window.event && window.event.clientY) ? window.event.clientY - rect.top : (canvas._lastMouseY || 0);
  // But to be robust, track mouse via listeners
  drawHover(canvas._mouseX || 0, canvas._mouseY || 0);
  drawWinnerLine();
  computeAndDrawArrows();
  updateUI();
  requestAnimationFrame(loop);
}
canvas.addEventListener('mousemove', (e)=> {
  const rect = canvas.getBoundingClientRect();
  canvas._mouseX = e.clientX - rect.left;
  canvas._mouseY = e.clientY - rect.top;
  canvas._lastMouseX = canvas._mouseX; canvas._lastMouseY = canvas._mouseY;
});

// initialize UI
updateUI();
requestAnimationFrame(loop);
</script>
